	{must1}{exp}0:不循环 1:循环{/exp}{int(0:1)}
[!LOOP]

	{must1}{exp}0:不使用 1:使用{/exp}{int(0:1)}
[!SHADOW]

	{must1}{exp}若为3，最后一帧必须以FRAME002结尾{/exp}{int(0:)}
[!FRAME MAX]

	{must1}{exp}0:不使用,1:使用{/exp}{int(0:1)}
[!SPECTRUM]

	{must1}{exp}1/1000秒{/exp}{int(0:)}
[!SPECTRUM TERM]

	{must1}{exp}1/1000秒{/exp}{int(0:)}
[!SPECTRUM LIFE TIME]

{exp}残影颜色{/exp}[SPECTRUM COLOR]
	{must1}{exp}{/exp}{int(0:)}
	{must2}{exp}{/exp}{int(0:)}
	{must3}{exp}{/exp}{int(0:)}
	{must4}{exp}{/exp}{int(0:)}

	{must1}{exp}NONE 或 DODGE 或 LINEARDODGE 或 DARK 或 XOR 或 MONOCHROME，其他会出错{/exp}{string}
[!SPECTRUM EFFECT]

	{must1}{exp}{/exp}{float(:)}

	{must1}{exp}红{/exp}{int(0:)}
	{must2}{exp}绿{/exp}{int(0:)}

	{must4}{exp}透明度{/exp}{int(0:)}

	{must1}{exp}0:不使用,1:使用{/exp}{int(0:1)}
[!INTERPOLATION]

	{must1}{exp}NONE 或 DODGE 或 LINEARDODGE 或 DARK 或 XOR 或 MONOCHROME{/exp}{string}
[!GRAPHIC EFFECT]

{exp}帧持续时间{/exp}[DELAY]

	{must1}{exp}NORMAL 或 SUPERARMOR 或 UNBREAKABLE{/exp}{string}
